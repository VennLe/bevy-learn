// src/animation/plugin.rs
use bevy::prelude::*;
use bevy::animation::AnimationPlayer;

pub struct CharacterAnimationPlugin;

impl Plugin for CharacterAnimationPlugin {
    fn build(&self, app: &mut App) {
        app.add_systems(Update, play_character_animations);
    }
}

// 播放 glTF 内嵌的动画
fn play_character_animations(
    mut player_query: Query<&mut AnimationPlayer, Added<AnimationPlayer>>,
) {
    for mut player in player_query.iter_mut() {
        // 播放第一个动画（通常是 idle）
        // 注意：glTF 中的动画索引从 0 开始
        if let Some(animation) = player.animation_clips().next() {
            player.play(animation).repeat();  // 循环播放
        }
    }
}

// 更精细的控制：选择特定动画
fn switch_animation(
    mut player_query: Query<&mut AnimationPlayer>,
    keyboard: Res<ButtonInput<KeyCode>>,
) {
    for mut player in player_query.iter_mut() {
        if keyboard.just_pressed(KeyCode::Space) {
            // 切换到第二个动画（假设是攻击动画）
            if let Some(attack_anim) = player.animation_clips().nth(1) {
                player.play(attack_anim).repeat();
            }
        }
    }
}

// 加载独立的动画文件（如果动画与模型分开存储）
fn load_separate_animation(
    mut commands: Commands,
    asset_server: Res<AssetServer>,
) {
    // 加载单独的动画剪辑
    let walk_anim: Handle<AnimationClip> = asset_server.load("animations/walk.glb#Animation0");
    let attack_anim: Handle<AnimationClip> = asset_server.load("animations/attack.glb#Animation0");
    
    // 创建动画图并添加多个动画
    let mut graph = AnimationGraph::new();
    let walk_node = graph.add_clip(walk_anim, 1.0, graph.root());
    let attack_node = graph.add_clip(attack_anim, 1.0, graph.root());
    
    let graph_handle = commands.register_asset(graph);
    
    commands.spawn((
        AnimationPlayer::default(),
        AnimationGraphHandle(graph_handle),
        Transform::default(),
    ));
}
